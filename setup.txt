#Install binaries
wget https://dev.mysql.com/get/Downloads/MySQL-8.4/mysql-8.4.7-linux-glibc2.28-x86_64.tar.xz
tar zxfv mysql-8.4.7-linux-glibc2.28-x86_64.tar.xz 
mv mysql-8.4.7-linux-glibc2.28-x86_64 mysql-8.4.7

#install prereq
sudo pacman -S ncurses
sudo ln -s /usr/lib/libncursesw.so.6.5 /usr/lib/libncurses.so.6
sudo pacman -S numactl
sudo pacman -S libaio

#test
$MSQL_DB1_HOME/bin/mysql --help
$MSQL_DB1_HOME/bin/mysqld --help

#initialize
$MSQL_DB1_HOME/bin/mysqld --initialize --datadir=/home/daddy/apps/mysql/db1/data
$MSQL_DB2_HOME/bin/mysqld --initialize --datadir=/home/daddy/apps/mysql/db2/data

#reset password
$MSQL_DB1_HOME/bin/mysqld_safe --defaults-file=$MSQL_DB1_HOME/config/my.cnf --skip-grant-tables &
#on another terminal and login with NO PASSWORD
$MSQL_DB1_HOME/bin/mysql --defaults-file=$MSQL_DB1_HOME/config/my.cnf -u root
#set to no password
UPDATE mysql.user SET authentication_string=null WHERE User='root';
FLUSH PRIVILEGES;
exit;
#login again
$MSQL_DB1_HOME/bin/mysql --defaults-file=$MSQL_DB1_HOME/config/my.cnf -u root
ALTER USER 'root'@'localhost' IDENTIFIED WITH caching_sha2_password BY 'password';
FLUSH PRIVILEGES;
exit;
